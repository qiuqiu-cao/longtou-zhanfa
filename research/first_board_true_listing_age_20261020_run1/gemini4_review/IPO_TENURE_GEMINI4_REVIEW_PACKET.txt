def listing_date_supported(date_count, official_source, listing_date, raw_dates):
    # Verification of name/ST intervals does not determine a separately
    # corroborated, stable IPO date. In particular, future interval states
    # cannot withhold a known historical listing date.
    return bool(date_count == 1 and official_source and len(raw_dates) == 1 and listing_date in raw_dates)


def derive_age(keys, identities):
    if keys.duplicated(KEYS).any() or identities.code.duplicated().any():
        raise ValueError("Duplicate identity or complete-pool key")
    out = keys.merge(identities, on="code", how="left", validate="many_to_one", sort=False)
    listed = pd.to_datetime(out.listing_date, format="%Y-%m-%d", errors="coerce")
    signal = pd.to_datetime(out.signal_date, format="%Y-%m-%d", errors="raise")
    valid = out.identity_qualified.fillna(False).astype(bool) & listed.notna() & listed.le(signal)
    out[NEW] = (signal - listed).dt.days.astype(float).where(valid)
    # This is calendar tenure. No pre-2023 official-session count is invented.
    return out

                             master_date_count=len(dates), master_source_official=bool(official),
                             all_security_state_intervals_verified=bool(declared),
                             master_row_pointers="|".join(str(i + 2) for i in part.index),
                             source_clock_original="|".join(sorted(part.source_timestamp.dropna().unique()))))
    identities = pd.DataFrame(identity)
    raw = []
    for path in RAW_LISTS:
        frame = pd.read_csv(path, dtype=str)
        if "A_STOCK_CODE" in frame:
            codes = frame.A_STOCK_CODE.str.zfill(6)
            dates = pd.to_datetime(frame.LIST_DATE, format="%Y%m%d", errors="raise").dt.strftime("%Y-%m-%d")
        else:
            codes, dates = frame.code.str.zfill(6), frame.listing_date
        for idx, code, listed in zip(frame.index, codes, dates):
            raw.append(dict(code=code, raw_listing_date=listed, pointer=f"{path.relative_to(base.ROOT).as_posix()}:csv_record={idx + 2}"))
    raw_frame = pd.DataFrame(raw)
    by_code = {code: part for code, part in raw_frame.groupby("code", sort=False)}
    match, conflicts, pointers = [], [], []
    for row in identities.itertuples():
        records = by_code.get(row.code, raw_frame.iloc[:0])
        dates = set(records.raw_listing_date.dropna())
        good = listing_date_supported(row.master_date_count, row.master_source_official, row.listing_date, dates)
        match.append(good)
        conflicts.append(bool(dates and (len(dates) != 1 or row.listing_date not in dates)))
        pointers.append("|".join(records.pointer))
    identities["identity_qualified"] = match
    identities["raw_date_conflict"] = conflicts
    identities["raw_row_pointers"] = pointers
    identities.to_csv(QUAL / "LISTING_DATE_IDENTITIES_WITH_RAW_POINTERS.csv", index=False, encoding="utf-8-sig")
    keys = pd.read_csv(SOURCE, dtype={"code": str}, usecols=KEYS + ["stage", OLD], float_precision="round_trip")
    keys = keys[keys.stage.eq(1)].reset_index(drop=True)
    output = derive_age(keys, identities[["code", "listing_date", "identity_qualified"]])
    for row in output[output[NEW].notna()].itertuples():
        assert getattr(row, NEW) == (date.fromisoformat(row.signal_date) - date.fromisoformat(row.listing_date)).days
    checks = []
    for cutoff in ("2025-06-30", "2025-12-31"):

def attach_qualified_age(rows, extra):
    columns = KEYS + [NEW]
    if rows.duplicated(KEYS).any() or extra.duplicated(KEYS).any():
        raise ValueError("Duplicate age feature keys")
    if set(map(tuple, rows[KEYS].to_numpy())) != set(map(tuple, extra[KEYS].to_numpy())):
        raise ValueError("Incomplete unchanged first-board denominator")
    out = rows.merge(extra[columns], on=KEYS, validate="one_to_one", sort=False)
    np.testing.assert_allclose(rows[ORIGINAL_FIELDS + [stage.TARGET]], out[ORIGINAL_FIELDS + [stage.TARGET]], equal_nan=True, rtol=0, atol=0)
    if out[NEW].dropna().lt(0).any():
        raise ValueError("Future listing date")
    return out

import numpy as np
import pandas as pd
import pytest

import first_board_true_listing_age as model


def test_ipo_date_support_has_no_dependency_on_other_security_state_intervals():
    assert model.listing_date_supported(1, True, "1999-05-27", {"1999-05-27"})
    assert not model.listing_date_supported(2, True, "1999-05-27", {"1999-05-27"})
    assert not model.listing_date_supported(1, False, "1999-05-27", {"1999-05-27"})
    assert not model.listing_date_supported(1, True, "1999-05-27", {"2000-01-01"})


def test_calendar_age_includes_pre_archive_history_and_leap_day():
    keys = pd.DataFrame({"signal_date": ["2024-03-01", "2024-03-01"], "code": ["000001", "000002"]})
    identity = pd.DataFrame({"code": ["000001", "000002"], "listing_date": ["1991-04-03", "2024-02-28"], "identity_qualified": [True, True]})
    result = model.derive_age(keys, identity)
    assert result.loc[0, model.NEW] > 12000
    assert result.loc[1, model.NEW] == 2
    assert model.OLD not in model.FIELDS and len(model.FIELDS) == 62
    assert set(model.FIELDS) - set(model.ORIGINAL_FIELDS) == {model.NEW}


def test_missing_conflicting_or_future_listing_date_remains_unknown():
    keys = pd.DataFrame({"signal_date": ["2024-01-02"] * 3, "code": ["000001", "000002", "000003"]})
    identity = pd.DataFrame({"code": ["000001", "000002"], "listing_date": ["2024-01-03", "2000-01-01"], "identity_qualified": [True, False]})
    assert model.derive_age(keys, identity)[model.NEW].isna().all()


def test_duplicate_identity_rejected_and_future_rows_do_not_change_past_age():
    keys = pd.DataFrame({"signal_date": ["2024-01-02", "2025-01-02"], "code": ["000001", "000001"]})
    identity = pd.DataFrame({"code": ["000001"], "listing_date": ["2000-01-01"], "identity_qualified": [True]})
    assert model.derive_age(keys, identity).loc[0, model.NEW] == model.derive_age(keys.iloc[:1], identity).loc[0, model.NEW]
    with pytest.raises(ValueError, match="Duplicate"):
        model.derive_age(keys, pd.concat([identity, identity]))


def test_old_feature_and_unknown_label_are_not_replaced_in_fact_table():
    rows = pd.DataFrame({"signal_date": ["2024-01-02"], "code": ["000001"],
                         **{field: [1.] for field in model.ORIGINAL_FIELDS}, model.stage.TARGET: [np.nan]})
    extra = pd.DataFrame({"signal_date": ["2024-01-02"], "code": ["000001"], model.NEW: [10000.]})
    result = model.attach_qualified_age(rows, extra)
    assert result.loc[0, model.OLD] == 1.
    assert np.isnan(result.loc[0, model.stage.TARGET])
    with pytest.raises(ValueError, match="Incomplete"):
        model.attach_qualified_age(rows, extra.iloc[:0])

"""Separate first-to-second and second-to-third EOD research models.

Only adjacent official-session closes define promotion. A second-board recheck
uses newly available second-board facts and never alters the first-board call.

       "turnover_ratio_5","prior_compression_5","two_session_turnover_sum",

def build_rows(inputs=base.INPUTS,feature_cutoff=None):
    first,bars,sessions,audit=base.build_dataset(inputs,feature_cutoff=feature_cutoff)
    g=bars.groupby("code",sort=False)
    continuous60=bars.p.sub(g.p.shift(60)).eq(60)
    bars["gene_max_streak_60"]=g.height.transform(lambda x:x.shift().rolling(60,min_periods=60).max()).where(continuous60)
    bars["gene_limits_60"]=g.limit.transform(lambda x:x.shift().rolling(60,min_periods=60).sum()).where(continuous60)
    historical_success=bars.limit&bars.previous_limit
    numerator=historical_success.groupby(bars.code).transform(lambda x:x.shift().rolling(60,min_periods=60).sum())
    denominator=g.previous_limit.transform(lambda x:x.shift().rolling(60,min_periods=60).sum())
    bars["gene_relay_rate_60"]=(numerator/denominator.where(denominator.gt(0))).where(continuous60)
    prior_turnover=g.turnover.transform(lambda x:x.shift().rolling(5,min_periods=5).mean())
    bars["turnover_ratio_5"]=(bars.turnover/prior_turnover.where(prior_turnover.gt(0))).where(bars.p.sub(g.p.shift(5)).eq(5))
    hi=g.high.transform(lambda x:x.shift().rolling(5,min_periods=5).max())
    lo=g.low.transform(lambda x:x.shift().rolling(5,min_periods=5).min())
    bars["prior_compression_5"]=(hi/lo-1).where(bars.p.sub(g.p.shift(5)).eq(5))
    adjacent=bars.p.sub(g.p.shift()).eq(1)
    bars["two_session_turnover_sum"]=(bars.turnover+g.turnover.shift()).where(adjacent)
    bars["previous_one_price"]=g.one_price.shift().where(adjacent)
    bars["opened_after_one_price"]=((bars.previous_one_price.eq(1))&bars.one_price.eq(0)).astype(float).where(bars.height.eq(2))
    # Counts are only complete within the declared archived security universe.
    eligible=bars[bars.eligible&bars.valid_price]
    market=eligible.groupby("date").height.max().to_frame("height")
    market["cycle_height_change"]=market.height.diff()
    cap=market.height.shift().rolling(20,min_periods=20).max()
    market["cycle_height_break_20"]=market.height.gt(cap).astype(float).where(cap.notna())
    anchors=bars.previous_height.ge(4)&bars.previous_limit
    for col,mask in [("cycle_anchor_breaks",anchors&~bars.limit),("cycle_anchor_collapses",anchors&bars.day_return.le(-.07))]:
        market[col]=bars[mask].groupby("date").size().reindex(market.index,fill_value=0)
    denominator=bars[anchors].groupby("date").size()
    numerator=bars[anchors&bars.limit].groupby("date").size()
    market["cycle_high_survival"]=numerator.reindex(market.index,fill_value=0)/denominator.reindex(market.index).replace(0,np.nan)
    first_count=eligible.groupby("date")["first"].sum()
    relay_count=eligible[eligible.height.ge(2)].groupby("date").size().reindex(market.index,fill_value=0)
    market["cycle_first_expansion_5"]=first_count/first_count.shift().rolling(5,min_periods=5).mean().replace(0,np.nan)
    market["cycle_first_relay_ratio"]=first_count/relay_count.replace(0,np.nan)
    top=eligible[eligible.height.eq(eligible.groupby("date").height.transform("max"))&eligible.height.ge(2)]
    market["cycle_top_one_price_share"]=top.groupby("date").one_price.mean()
    second=bars.height.eq(2)&bars.previous_limit&bars.previous_height.eq(1)&g["first"].shift().fillna(False).astype(bool)&adjacent
    rows=bars[bars["first"]|second].copy().rename(columns={"date":"signal_date"})
    rows["stage"]=np.where(rows["first"],1,2)
    rows=rows.merge(first[["signal_date"]+MARKET].drop_duplicates("signal_date"),on="signal_date",how="left",validate="many_to_one")
    rows=rows.merge(market.drop(columns="height"),left_on="signal_date",right_index=True,how="left",validate="many_to_one")
    for out,col in [("turnover_rank","turnover"),("amount_rank","amount"),("volume_ratio_rank","volume_ratio_5"),("pre_momentum_rank","return_20")]:
        rows[out]=rows.groupby(["signal_date","stage"])[col].rank(pct=True)
    rows["stage_pool_size"]=rows.groupby(["signal_date","stage"]).code.transform("size")
    rows["stage_one_price_share"]=rows.groupby(["signal_date","stage"]).one_price.transform("mean")
    lookup=bars.set_index(["p","code"])[["limit","valid_price","regular_price","eligible"]]
    idx=pd.MultiIndex.from_arrays([rows.p.to_numpy()+1,rows.code.to_numpy()],names=["p","code"])
    future=lookup.reindex(idx)
    observed=future.valid_price.fillna(False).to_numpy(dtype=bool)&future.regular_price.fillna(False).to_numpy(dtype=bool)&future.eligible.fillna(False).to_numpy(dtype=bool)
    rows[TARGET]=np.where(observed,future.limit.fillna(False).to_numpy(dtype=float),np.nan)
    rows["label_end"]=rows.p.map(lambda p:sessions[p+1] if p+1<len(sessions) else None)
    rows["baseline_score"]=-rows.turnover
    fields=VARIANTS["STAGE_CYCLE_GENE"]
    rows[fields]=rows[fields].replace([np.inf,-np.inf],np.nan)
    rows=rows.sort_values(["signal_date","stage","code"]).reset_index(drop=True)
    audit.update(stage_rows=rows.groupby("stage").size().to_dict(),stage_known=rows.groupby("stage")[TARGET].count().to_dict(),
                 next_session_label="Exactly next official session; next session missing/ineligible/invalid price remains UNKNOWN",extra_features=EXTRA,
                 stage2_identity="Previous adjacent official session is a verified reconstructed first board, current close is second exact limit")
    return rows,bars,sessions,audit


def metric(rows,score="model_score",k=1):
    take=base.selected(rows,score,k)
    known=take[TARGET].notna();hits=int(take[TARGET].eq(1).sum());n=int(known.sum())
    total_positive=int(rows[TARGET].eq(1).sum())
    if n:
        z=1.959963984540054;p=hits/n;den=1+z*z/n
        center=(p+z*z/(2*n))/den;half=z*math.sqrt(p*(1-p)/n+z*z/(4*n*n))/den
        interval=[max(0,center-half),min(1,center+half)]
    else:interval=[None,None]
    return dict(selected=len(take),selected_dates=take.signal_date.nunique(),known=n,unknown=int((~known).sum()),hits=hits,
                hit_rate=hits/n if n else None,wilson95=interval,full_positive_recall=hits/total_positive if total_positive else None,
                false_positives=int(take[TARGET].eq(0).sum()),false_negatives=total_positive-hits,positive_pool=total_positive,
                outcome_coverage=n/len(take) if len(take) else None)


def calibration_metrics(rows):
    known=rows[TARGET].notna()&rows.calibrated_score.notna()
    x=rows[known];y=x[TARGET].astype(int);p=x.calibrated_score
    if not len(x):return dict(status="UNAVAILABLE",historical_gate_passed=False)
    groups=pd.cut(p,np.linspace(0,1,11),include_lowest=True)
    bins=x.assign(bin=groups).groupby("bin",observed=True).agg(n=(TARGET,"size"),mean_probability=("calibrated_score","mean"),observed=(TARGET,"mean"))
    ece=float((bins.n*(bins.mean_probability-bins.observed).abs()).sum()/len(x))
    brier=brier_score_loss(y,p);baseline=brier_score_loss(y,x.train_prior)
    top=base.selected(rows,"model_score",1);top=top[top[TARGET].notna()&top.calibrated_score.notna()]
    top_error=abs(float(top.calibrated_score.mean()-top[TARGET].mean())) if len(top) else None
    qualified=bool(len(x)>=500 and len(top)>=200 and brier<baseline and ece<=.08 and top_error<=.10)
    return dict(status="HISTORICAL_TEMPORAL_CALIBRATION_ONLY",historical_gate_passed=qualified,rows=len(x),brier=brier,prior_brier=baseline,ece=ece,
                top1_rows=len(top),top1_forecast_mean=float(top.calibrated_score.mean()),top1_observed=float(top[TARGET].mean()),top1_absolute_gap=top_error,
                bins=[dict(bin=str(idx),**row.to_dict()) for idx,row in bins.iterrows()])


def evaluate(rows):
    known=rows[TARGET].notna()
    y=rows.loc[known,TARGET].astype(int);score=rows.loc[known,"model_score"]
    p,r,f,_=precision_recall_fscore_support(y,score.ge(.5),average="binary",zero_division=0)
    return dict(rows=len(rows),sessions=rows.signal_date.nunique(),known=int(known.sum()),unknown=int((~known).sum()),pool_rate=float(y.mean()),
                top={str(k):metric(rows,k=k) for k in (1,3,5)},baseline_top={str(k):metric(rows,score="baseline_score",k=k) for k in (1,3,5)},
                threshold_0_5_uncalibrated=dict(accuracy=float((score.ge(.5).to_numpy()==y.to_numpy()).mean()),precision=p,recall=r,f1=f,auc=roc_auc_score(y,score)),
                calibration=calibration_metrics(rows))


def fit_core(train,fields,out,name):
    directory=out/"models";directory.mkdir(exist_ok=True)
    receipt=dict(name=name,started_at=base.now(),rows=len(train),positives=int(train[TARGET].sum()),max_label_end=train.label_end.max(),
                 min_signal=train.signal_date.min(),max_signal=train.signal_date.max(),features=fields)
    before=directory/f"{name}_BEFORE_FIT.json"
    if before.exists():before=directory/f"{name}_BEFORE_FIT_REPAIR.json"
    observed=[f for f in fields if train[f].notna().any()]
    receipt["all_missing_columns_dropped_from_estimator_only"]=sorted(set(fields)-set(observed))
    base.write_json(before,receipt)
    train[["signal_date","code","stage","label_end",TARGET]].to_csv(directory/f"{name}_members.csv",index=False)
    # sklearn1.9's histogram binning rejects a completely empty feature.
    # Learn the no-information column set from this training window only;
    # retain original NaNs in all fact/feature files and do not invent values.
    model=Pipeline([("observed_columns",ColumnTransformer([("observed","passthrough",observed)],remainder="drop")),("hist",base.make_model("hist_gradient"))])
    weights=1/train.groupby("signal_date").code.transform("size");weights=weights/weights.mean()
    model.fit(train[fields],train[TARGET].astype(int),hist__sample_weight=weights)
    joblib.dump(model,directory/f"{name}.joblib")
    receipt.update(completed_at=base.now(),sha256=base.sha(directory/f"{name}.joblib"))
    base.write_json(directory/f"{name}_COMPLETED.json",receipt)
    return model


def predict(bundle,score):
    out=score.copy();raw=bundle["model"].predict_proba(out[bundle["features"]])[:,1]
    out["model_score"]=raw
    if bundle["calibrator"] is not None:
        logits=np.log(np.clip(raw,1e-6,1-1e-6)/np.clip(1-raw,1e-6,1-1e-6))
        out["calibrated_score"]=bundle["calibrator"].predict_proba(logits.reshape(-1,1))[:,1]
    else:out["calibrated_score"]=np.nan
    out["train_prior"]=bundle["train_prior"]
    return out


def fit_head(train,variant,out,name):
    fields=VARIANTS[variant];dates=sorted(train.signal_date.unique())
    cached=out/"models"/f"{name}_bundle.joblib"
    if cached.exists():
        bundle=joblib.load(cached)
        member=pd.read_csv(out/"models"/f"{name}_full_members.csv",dtype={"code":str})
        assert set(map(tuple,member[["signal_date","code","stage"]].to_numpy()))==set(map(tuple,train[["signal_date","code","stage"]].to_numpy()))
        assert bundle["features"]==fields and bundle["train_max_label_end"]==train.label_end.max()
        receipt=json.loads((out/"models"/f"{name}_full_COMPLETED.json").read_text("utf-8"))
        assert base.sha(out/"models"/f"{name}_full.joblib")==receipt["sha256"]
        base.write_json(out/"models"/f"{name}_REUSED_WITHOUT_FIT.json",dict(reused_at=base.now(),bundle_sha256=base.sha(cached),original_completed=receipt["completed_at"]))
        return bundle
    if len(dates)<100:raise ValueError("insufficient temporal training sessions")
    cutoff=dates[-42];inner=train[train.label_end.lt(cutoff)];cal=train[train.signal_date.ge(cutoff)]
    inner_model=fit_core(inner,fields,out,name+"_calibration_prefix")
    raw=inner_model.predict_proba(cal[fields])[:,1]
    logit=np.log(np.clip(raw,1e-6,1-1e-6)/np.clip(1-raw,1e-6,1-1e-6))
    calibrator=LogisticRegression(C=1.,max_iter=1000,random_state=42).fit(logit.reshape(-1,1),cal[TARGET].astype(int))
    if calibrator.coef_[0,0]<=0:calibrator=None
    cal_receipt=dict(name=name,cutoff=cutoff,inner_max_label_end=inner.label_end.max(),calibration_rows=len(cal),calibration_range=[cal.signal_date.min(),cal.signal_date.max()],
                     positive_slope=calibrator is not None,formed_at=base.now())
    cal[["signal_date","code","stage","label_end",TARGET]].assign(uncalibrated_oof=raw).to_csv(out/"models"/f"{name}_calibration_rows.csv",index=False)
    base.write_json(out/"models"/f"{name}_calibration_receipt.json",cal_receipt)
    model=fit_core(train,fields,out,name+"_full")
    bundle=dict(version=VERSION,stage=int(train.stage.iloc[0]),model=model,calibrator=calibrator,features=fields,variant=variant,
                train_prior=float(train[TARGET].mean()),train_max_label_end=train.label_end.max(),max_signal=train.signal_date.max(),
                calibration_status="HISTORICAL_QUALIFICATION_PENDING",research_only=True)
    joblib.dump(bundle,out/"models"/f"{name}_bundle.joblib")
    return bundle


def chain_recheck(first_predictions,second_predictions):
    picked=base.selected(first_predictions,"model_score",1).copy()
    picked=picked.rename(columns={"signal_date":"first_signal_date",TARGET:"first_promotion_LABEL_ONLY","model_score":"first_model_score","calibrated_score":"first_calibrated_score","label_end":"second_signal_date"})
    second=second_predictions[["signal_date","code","model_score","calibrated_score",TARGET,"label_end"]].rename(columns={"signal_date":"second_signal_date","model_score":"second_model_score","calibrated_score":"second_calibrated_score",TARGET:"second_promotion_LABEL_ONLY","label_end":"third_target_date"})
    keep=["first_signal_date","code","name","second_signal_date","first_promotion_LABEL_ONLY","first_model_score","first_calibrated_score"]
    merged=picked[keep].merge(second,on=["second_signal_date","code"],how="left",validate="one_to_one")
    # Second-board assessment exists only when the first call actually promoted.
    fields=["second_model_score","second_calibrated_score","second_promotion_LABEL_ONLY","third_target_date"]
    merged.loc[~merged.first_promotion_LABEL_ONLY.eq(1),fields]=np.nan

    continued=merged.second_promotion_LABEL_ONLY.eq(1)&merged.first_promotion_LABEL_ONLY.eq(1)
    qualified=matured&merged.second_calibrated_score.ge(.5)
    stats=dict(first_selections=len(merged),first_promoted=int(merged.first_promotion_LABEL_ONLY.eq(1).sum()),second_rechecks_available=int(merged.second_model_score.notna().sum()),
               second_outcomes_known=int(matured.sum()),continued_to_third=int(continued.sum()),
               conditional_second_to_third=float(continued.sum()/matured.sum()) if matured.any() else None,
               qualified_rechecks=int(qualified.sum()),qualified_wins=int((qualified&continued).sum()),
               qualified_hit_rate=float((qualified&continued).sum()/qualified.sum()) if qualified.any() else None,
               full_first_to_third_rate=float(continued.sum()/merged.first_promotion_LABEL_ONLY.notna().sum()))
    return merged,stats

def recheck(bundle_path,features_path,output):

    if not rows.stage.isin([1,2]).all():raise ValueError("stage must be1 or2")
    parts=[]
    for stage,part in rows.groupby("stage"):
        head=bundle["heads"][int(stage)]
        if set(head["features"])-set(part):raise ValueError("incomplete features")

        scored=predict(head,part)
        scored["probability_status"]=head["calibration_status"]
        if head["calibration_status"]!="HISTORICAL_QUALIFIED":scored["calibrated_score"]=np.nan
        parts.append(scored)
    result=pd.concat(parts,ignore_index=True)
    result=result[[c for c in result if "LABEL_ONLY" not in c and c!="label_end"]]

def qualified_preclose_evidence(rows, records, strict_acquisition=False):
    """Bind author-reviewed original evidence, rejecting post-close revelations."""
    keys = set(zip(rows.signal_date, rows.code))
    accepted = {}
    for record in records:
        key = (record["signal_date"], record["code"])
        if key not in keys:
            raise ValueError("Evidence is outside the complete signal cohort")
        cutoff = pd.Timestamp(key[0] + "T15:00:00+08:00")
        published = pd.Timestamp(record["published_at"])
        acquired = pd.Timestamp(record["acquired_at"])
        if published.tzinfo is None or acquired.tzinfo is None:
            raise ValueError("Evidence timestamps require explicit timezones")
        if published > cutoff or (strict_acquisition and acquired > cutoff):
            raise ValueError("Evidence was not available at the signal close")
        if acquired < published:
            raise ValueError("Acquisition cannot precede publication")
        if record.get("review_status") != "AUTHOR_VERIFIED" or not record.get("rationale"):
            raise ValueError("Missing independent source review")
        raw = Path(record["raw_path"])
        if not raw.is_file() or base.sha(raw) != record["raw_sha256"]:
            raise ValueError("Raw evidence hash mismatch")
        kind = record["kind"]
        if kind == "RESTRUCTURING_DISCLOSURE":
            domain = (urlparse(record["source_url"]).hostname or "").lower()
            if not any(domain == d or domain.endswith("." + d) for d in ("cninfo.com.cn", "sse.com.cn", "szse.cn")):
                raise ValueError("Restructuring exception needs an official disclosure")
            field = "verified_restructuring_preclose"
        elif kind == "SUSPECTED_THREEBOARD_STRUCTURE":
            channels = set(record.get("independently_verified_channels", []))
            if len(channels & {"minute_abrupt_pull", "coordinated_seats", "repeated_order_cancellations", "verified_weak_sector"}) < 2:
                raise ValueError("Single-seat or daily-bar pattern cannot establish suspected fund structure")
            field = "verified_threeboard_risk_preclose"
        else:
            raise ValueError("Unsupported evidence kind")
        accepted.setdefault(key, {})[field] = 1.
    return accepted


def capital_context(rows, upper_cap, evidence=None, strict_acquisition=False):
    """Only current/past columns and same-date complete cohort enter features."""
    out = rows.copy()
    cap = pd.to_numeric(out.float_cap, errors="coerce").where(lambda x: x.gt(0))
    out["cap_rank_in_first_pool"] = cap.groupby(out.signal_date).rank(pct=True)
    median = cap.groupby(out.signal_date).transform("median")
    out["log_cap_to_first_pool_median"] = np.log(cap / median)
    out["log_cap_excess_development80"] = np.log(cap / upper_cap).clip(lower=0)
    small = cap.le(RISK_POLICY["microcap_yuan"]).astype(float).where(cap.notna())
    low = out.turnover.lt(RISK_POLICY["low_turnover_percent"]).astype(float).where(out.turnover.notna())
    compressed = out.volume_ratio_5.lt(RISK_POLICY["volume_ratio_max"]).astype(float).where(out.volume_ratio_5.notna())
    out["microcap_low_turnover_proxy"] = small * low
    out["microcap_one_price_proxy"] = small * out.one_price
    out["microcap_volume_compression_proxy"] = small * compressed
    # These unavailable evidence fields remain unknown, never zero-filled.
    for f in ("verified_restructuring_preclose", "verified_threeboard_risk_preclose",
              "abrupt_first_pull_verified", "coordinated_seats_verified", "weak_sector_verified"):
        if f in out and out[f].notna().any():
            raise ValueError("Evidence flags must come from dated reviewed raw evidence, not feature-file assertions")
        out[f] = np.nan
    accepted = qualified_preclose_evidence(out, evidence or [], strict_acquisition)
    for i, key in zip(out.index, zip(out.signal_date, out.code)):
        for field, value in accepted.get(key, {}).items():
            out.loc[i, field] = value
    out["threeboard_evidence_status"] = np.where(
        out.verified_threeboard_risk_preclose.eq(1), "SUSPECTED_WITH_DATED_EVIDENCE", "INSUFFICIENT_EVIDENCE")
    # Proxy-only exclusions are separately named and never relabelled as fund identities.
    thin = out.microcap_low_turnover_proxy.eq(1) & (out.one_price.eq(1) | compressed.eq(1))
    large = cap.gt(upper_cap) & ~out.verified_restructuring_preclose.eq(1)
    confirmed_risk = out.verified_threeboard_risk_preclose.eq(1)
    out["capital_missing_review"] = cap.isna()
    out["capital_oversize_review"] = large
    out["thin_turnover_proxy_excluded"] = thin
    out["threeboard_evidence_excluded"] = confirmed_risk
    out["research_selection_allowed"] = ~(cap.isna() | large | thin | confirmed_risk)
    out["exclusion_reason"] = ["|".join(label for flag, label in zip(flags, (
        "CAP_MISSING_REVIEW", "ABOVE_DEVELOPMENT_CAP_NO_VERIFIED_EXCEPTION",
        "CONSERVATIVE_MICROCAP_THIN_TURNOVER_PROXY", "DATED_THREEBOARD_RISK")) if flag)
        for flags in zip(cap.isna(), large, thin, confirmed_risk)]
    return out


def screen_metric(rows, score="model_score"):
    allowed = rows[rows.research_selection_allowed]
    m = stage.metric(allowed, score=score)
    # Recall uses the original complete positive pool, including excluded winners.
    total_positive = int(rows[stage.TARGET].eq(1).sum())
    m.update(full_positive_pool=total_positive,
             full_positive_recall=m["hits"] / total_positive if total_positive else None,
             false_negatives=total_positive - m["hits"],
             full_signal_dates=int(rows.signal_date.nunique()),
             abstention_dates=int(rows.signal_date.nunique() - allowed.signal_date.nunique()),
             decision_date_coverage=float(allowed.signal_date.nunique() / rows.signal_date.nunique()),
             successful_predictions_per_all_pool_dates=float(m["hits"] / rows.signal_date.nunique()),
             excluded_rows=int((~rows.research_selection_allowed).sum()),
             excluded_known_promoters=int((~rows.research_selection_allowed & rows[stage.TARGET].eq(1)).sum()),
             excluded_unknown=int((~rows.research_selection_allowed & rows[stage.TARGET].isna()).sum()))
    return m


def prefix_checks(rows, upper_cap):
    checks = []
    expected = capital_context(rows, upper_cap)
    for cutoff in ("2025-06-30", "2025-12-31"):
        p = capital_context(rows[rows.signal_date.le(cutoff)].drop(columns=[stage.TARGET, "label_end"]), upper_cap)
        old = expected[expected.signal_date.le(cutoff)]
        np.testing.assert_allclose(p[CAP_FIELDS], old[CAP_FIELDS], equal_nan=True, rtol=1e-12, atol=1e-12)
        assert p.research_selection_allowed.tolist() == old.research_selection_allowed.tolist()
        checks.append(dict(cutoff=cutoff, rows=len(p), all_six_fields_and_screen_equal=True, targets_absent=True))
    return checks


def load_rows():
    path = REFERENCE / "all_first_and_second_boards_WITH_SEPARATE_LABELS.csv"
    rows = pd.read_csv(path, dtype={"code": str}, float_precision="round_trip")
    rows = rows[rows.stage.eq(1)].copy()
    names = pd.read_csv(SYMBOLIC / "complete28503_first_board_dated_symbolic_features.csv", dtype={"code": str}, float_precision="round_trip")
    names = names.rename(columns={"date": "signal_date"})
    assert set(zip(rows.signal_date, rows.code)) == set(zip(names.signal_date, names.code))
    rows = rows.merge(names[["signal_date", "code"] + symbolic.FIELDS], on=["signal_date", "code"], validate="one_to_one")
    development = rows[rows.signal_date.lt("2025-01-01") & rows.label_end.lt("2025-01-01") & rows[stage.TARGET].notna()]
    cutoff = sorted(development.signal_date.unique())[-42]
    scale_train = development[development.label_end.lt(cutoff)]
    upper_cap = float(scale_train.float_cap.where(scale_train.float_cap.gt(0)).quantile(RISK_POLICY["upper_cap_quantile"]))
    return rows, upper_cap, dict(max_scale_signal=scale_train.signal_date.max(), max_scale_label_end=scale_train.label_end.max(),
                                 calibration_start=cutoff, rows=len(scale_train), upper_float_cap_yuan=upper_cap)


def attach_frozen(rows, file):
    ref = pd.read_csv(file, dtype={"code": str}, float_precision="round_trip")
    part = rows[rows.signal_date.isin(ref.signal_date.unique())].copy()
    assert set(zip(part.signal_date, part.code)) == set(zip(ref.signal_date, ref.code))
    old = part.set_index(["signal_date", "code"])[stage.TARGET].sort_index()
    new = ref.set_index(["signal_date", "code"])[stage.TARGET].sort_index()
    np.testing.assert_allclose(old, new, equal_nan=True, atol=0, rtol=0)
    return part.merge(ref[["signal_date", "code", "model_score", "calibrated_score", "train_prior"]],
                      on=["signal_date", "code"], validate="one_to_one")


def cap_distribution(rows, out):
    # Descriptive bins are fixed before any outcome inspection; not new filter candidates.
    binned = rows.assign(float_cap_bin=pd.cut(rows.float_cap / 1e8, [0, 20, 30, 50, 100, 200, 500, np.inf],
                    labels=["<20", "20-30", "30-50", "50-100", "100-200", "200-500", ">=500"]))
    binned["float_cap_bin"] = binned.float_cap_bin.astype(object).fillna("UNKNOWN")
    binned["period"] = np.select([binned.signal_date.lt("2025-01-01"), binned.signal_date.lt("2025-07-01"), binned.signal_date.lt("2026-01-01")],
                                  ["development", "validation2025H1", "evaluation2025H2"], default="evaluation2026")
    result = binned.groupby(["period", "float_cap_bin"], sort=True).agg(rows=("code", "size"), known=(stage.TARGET, "count"), hits=(stage.TARGET, "sum"))
    result["hit_rate"] = result.hits / result.known
    result.reset_index().to_csv(out / "fixed_cap_bins_DESCRIPTIVE_NOT_THRESHOLD_SEARCH.csv", index=False)
    return result.reset_index().to_dict("records")

import pandas as pd
import numpy as np
import hashlib

def sha256(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()

f1 = 'AUDIT_COMPLETE4724_SCORE_COMPARISON_WITH_UNKNOWN.csv'
f2 = 'INDEPENDENT_ALL117_DAILY_PAIRED_SELECTIONS.csv'
print("SHA256 f1:", sha256(f1))
print("SHA256 f2:", sha256(f2))

df = pd.read_csv(f1, dtype={'code': str})
pairs = pd.read_csv(f2, dtype={'challenger_code': str, 'baseline_code': str})

print("df shape:", df.shape)
print("df dates:", df['signal_date'].nunique())
print("known labels:", df['promotion_next_LABEL_ONLY'].notna().sum(), "unknown:", df['promotion_next_LABEL_ONLY'].isna().sum())
print("positives:", (df['promotion_next_LABEL_ONLY'] == 1).sum())
print("excluded rows:", (~df['research_selection_allowed']).sum())
print("excluded known promoters:", (~df['research_selection_allowed'] & (df['promotion_next_LABEL_ONLY'] == 1)).sum())
print("excluded unknown:", (~df['research_selection_allowed'] & df['promotion_next_LABEL_ONLY'].isna()).sum())

# Recompute selections
for non_one, scope in [(False, 'primary'), (True, 'non_one_price')]:
    mask = df['research_selection_allowed'] & ((df['one_price'] == 0) if non_one else True)
    sub = df[mask]
    c_sel = sub.sort_values(['signal_date', 'model_score', 'code'], ascending=[True, False, True]).groupby('signal_date').head(1).set_index('signal_date')
    b_sel = sub.sort_values(['signal_date', 'baseline_raw_score', 'code'], ascending=[True, False, True]).groupby('signal_date').head(1).set_index('signal_date')
    
    c_hits = (c_sel['promotion_next_LABEL_ONLY'] == 1).sum()
    b_hits = (b_sel['promotion_next_LABEL_ONLY'] == 1).sum()
    c_known = c_sel['promotion_next_LABEL_ONLY'].notna().sum()
    b_known = b_sel['promotion_next_LABEL_ONLY'].notna().sum()
    
    ah = c_sel['promotion_next_LABEL_ONLY'] == 1
    bh = b_sel['promotion_next_LABEL_ONLY'] == 1
    common_wins = (ah & bh).sum()
    c_only = (ah & ~bh).sum()
    b_only = (~ah & bh).sum()
    common_fail = (~ah & ~bh).sum()
    
    print(f"[{scope}] dates={len(c_sel)}, c_hits={c_hits}/{c_known} ({c_hits/c_known:.4%}), b_hits={b_hits}/{b_known} ({b_hits/b_known:.4%})")
    print(f"   common_wins={common_wins}, c_only={c_only}, b_only={b_only}, common_fail={common_fail}")
    
    # Check against f2
    p_sub = pairs[pairs['scope'] == scope].set_index('date')
    assert (c_sel['code'] == p_sub['challenger_code']).all()
    assert (b_sel['code'] == p_sub['baseline_code']).all()
    print(f"   Matches {f2} exactly!")

# Unfiltered challenger
c_unfilt = df.sort_values(['signal_date', 'model_score', 'code'], ascending=[True, False, True]).groupby('signal_date').head(1)
print("Unfiltered challenger hits:", (c_unfilt['promotion_next_LABEL_ONLY'] == 1).sum(), "/", c_unfilt['promotion_next_LABEL_ONLY'].notna().sum())

# Check age columns
print("listing_calendar_age_days nulls:", df['listing_calendar_age_days'].isna().sum(), "min:", df['listing_calendar_age_days'].min(), "max:", df['listing_calendar_age_days'].max())
print("archive_age_groups_with_multiple_actual_tenures:", (df.groupby(['signal_date', 'listing_age_sessions'])['listing_calendar_age_days'].nunique() > 1).sum())
print("distinct_actual_age_values:", df['listing_calendar_age_days'].nunique())

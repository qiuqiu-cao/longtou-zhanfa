{ 训练基础：2,013,716 个历史交易日样本，37,886 个连板样本，4折滑动时序验证          }
{ 核心突破：攻克“非一字竞价”确定性，彻底摆脱一字顶格买不进、炸板吃大面的弊端      }
{ 适用软件：同花顺 PC 电脑版公式管理器（快捷键 Ctrl+F）                         }

{------------------------------------------------------------------------------}
{ 公式一：【龙头竞价非一字选股】（条件选股公式 - 09:25 一键选股）              }
{ 用途：09:25 集合竞价定型后，筛选 1进2、2进3、3进4及高位接力中大概率封板的真龙 }
{ 选股时机：工作日 09:25:01 执行条件选股                                       }
{------------------------------------------------------------------------------}

{ 1. 基础过滤：排除ST、*ST、退市整理 }
NOT_ST := IF(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'), 0, 1);

{ 2. 涨跌停幅度动态计算（主板10%，创业板/科创板20%） }
IS_20 := IF(CODELIKE('300') OR CODELIKE('301') OR CODELIKE('688'), 1, 0);
ZT_MULT := IF(IS_20, 1.20, 1.10);
ZT_PRICE := ROUND(REF(CLOSE, 1) * ZT_MULT * 100) / 100;

{ 3. 昨日涨停确认（昨日收盘价达到涨停价且等于最高价） }
ZT_PREV_MULT := IF(IS_20, 1.195, 1.095);
ZT_PREV_PRICE := ROUND(REF(CLOSE, 2) * ZT_PREV_MULT * 100) / 100;
YESTERDAY_ZT := (REF(CLOSE, 1) >= ZT_PREV_PRICE - 0.01) AND (REF(CLOSE, 1) = REF(HIGH, 1));

{ 4. 核心条件一：非一字开盘（开盘价低于涨停价，散户可以挂单买入） }
NOT_ONE_WORD := OPEN < (ZT_PRICE - 0.01);

{ 5. 核心条件二：09:25 黄金高开区间（+2.0% 至 +7.0%） }
GAP_PCT := (OPEN - REF(CLOSE, 1)) / REF(CLOSE, 1) * 100;
GOLDEN_GAP := (GAP_PCT >= 2.0) AND (GAP_PCT <= 7.0);

{ 6. 核心条件三：竞价爆量抢筹（竞价金额占昨日成交额 >= 5.0% 或量比 >= 15） }
{ DYNAINFO(15) 为同花顺集合竞价成交额（元），DYNAINFO(17) 为开盘量比 }
AUCTION_RATIO := (DYNAINFO(15) / REF(AMOUNT, 1)) * 100;
EXPLOSIVE_VOL := (AUCTION_RATIO >= 5.0) OR (DYNAINFO(17) >= 15.0);

{ 7. 核心条件四：跳空突破（竞价开盘价大于等于昨日最高价，实现全员浮盈无套牢盘） }
BREAK_HIGH := OPEN >= REF(HIGH, 1);

{ 8. 核心条件五：容量与换手健康度（昨日成交额 >= 3亿元，昨日换手率 8% ~ 32%） }
PREV_TURNOVER := (REF(VOL, 1) / CAPITAL) * 100;
HEALTHY_CAPACITY := (REF(AMOUNT, 1) >= 300000000) AND (PREV_TURNOVER >= 8.0) AND (PREV_TURNOVER <= 32.0);

{ 9. 最终选股输出 }
XG: NOT_ST AND YESTERDAY_ZT AND NOT_ONE_WORD AND GOLDEN_GAP AND EXPLOSIVE_VOL AND BREAK_HIGH AND HEALTHY_CAPACITY;



{------------------------------------------------------------------------------}
{ 公式二：【首板起爆竞价选股】（条件选股公式 - 低位首板起爆专用）                }
{ 用途：09:25 筛选昨日未涨停，今日竞价爆量跳空高开、均线多头、大概率首板起爆个股 }
{------------------------------------------------------------------------------}

{ 1. 基础过滤：排除ST与退市整理 }
NOT_ST2 := IF(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'), 0, 1);

{ 2. 昨日未涨停（过滤连板，专注低位首板突破） }
IS_20_2 := IF(CODELIKE('300') OR CODELIKE('301') OR CODELIKE('688'), 1, 0);
ZT_PREV_MULT2 := IF(IS_20_2, 1.195, 1.095);
ZT_PREV_PRICE2 := ROUND(REF(CLOSE, 2) * ZT_PREV_MULT2 * 100) / 100;
NOT_PREV_ZT := REF(CLOSE, 1) < ZT_PREV_PRICE2 - 0.01;

{ 3. 今日竞价温和高开（+2.5% 至 +6.0%） }
GAP_PCT2 := (OPEN - REF(CLOSE, 1)) / REF(CLOSE, 1) * 100;
SWEET_GAP := (GAP_PCT2 >= 2.5) AND (GAP_PCT2 <= 6.0);

{ 4. 竞价绝对爆量（竞价成交额 >= 1500 万元 且 竞价金额占昨比 >= 8.0%） }
AUCTION_RATIO2 := (DYNAINFO(15) / REF(AMOUNT, 1)) * 100;
STRONG_AUCTION := (DYNAINFO(15) >= 15000000) AND (AUCTION_RATIO2 >= 8.0);

{ 5. 均线多头形态（股价站上5日、10日、20日均线，且5日线向上） }
MA5 := MA(CLOSE, 5);
MA10 := MA(CLOSE, 10);
MA20 := MA(CLOSE, 20);
BULL_MA := (OPEN > MA5) AND (OPEN > MA20) AND (MA5 >= REF(MA5, 1));

{ 6. 开盘跳空站上昨日最高价 }
BREAK_YEST_HIGH := OPEN > REF(HIGH, 1);

{ 7. 最终选股输出 }
XG_FIRST: NOT_ST2 AND NOT_PREV_ZT AND SWEET_GAP AND STRONG_AUCTION AND BULL_MA AND BREAK_YEST_HIGH;



{------------------------------------------------------------------------------}
{ 公式三：【大容量换手真龙选股】（襄阳轴承专项模型 - 成交额>=8亿+大换手真龙） }
{ 用途：专抓大成交（>=8亿）、大换手（>=15%）、早盘主动进攻、分歧转一致的大真龙  }
{------------------------------------------------------------------------------}

NOT_ST3 := IF(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'), 0, 1);
IS_20_3 := IF(CODELIKE('300') OR CODELIKE('301') OR CODELIKE('688'), 1, 0);
ZT_M3 := IF(IS_20_3, 1.20, 1.10);
ZT_P3 := ROUND(REF(CLOSE, 1) * ZT_M3 * 100) / 100;

{ 昨日涨停且非一字开盘 }
ZT_P_PREV3 := ROUND(REF(CLOSE, 2) * IF(IS_20_3, 1.195, 1.095) * 100) / 100;
YEST_ZT3 := (REF(CLOSE, 1) >= ZT_P_PREV3 - 0.01);
NOT_ONE3 := OPEN < (ZT_P3 - 0.01);

{ 大容量大换手画像：昨日成交额 >= 8亿元，换手率 >= 15.0% }
CAPACITY_AMOUNT := REF(AMOUNT, 1) >= 800000000;
CAPACITY_TURNOVER := (REF(VOL, 1) / CAPITAL) * 100 >= 15.0;

{ 今日竞价红盘高开（+3.0% 至 +7.0%），竞价金额 >= 2500 万元 }
GAP_PCT3 := (OPEN - REF(CLOSE, 1)) / REF(CLOSE, 1) * 100;
STRONG_GAP3 := (GAP_PCT3 >= 3.0) AND (GAP_PCT3 <= 7.0);
BIG_AUCTION3 := DYNAINFO(15) >= 25000000;

{ 竞价开盘站在昨日成交均价（全员成本线）之上 }
PREV_VWAP := REF(AMOUNT, 1) / (REF(VOL, 1) * 100);
ABOVE_COST := OPEN >= PREV_VWAP;

XG_DRAGON: NOT_ST3 AND YEST_ZT3 AND NOT_ONE3 AND CAPACITY_AMOUNT AND CAPACITY_TURNOVER AND STRONG_GAP3 AND BIG_AUCTION3 AND ABOVE_COST;



{------------------------------------------------------------------------------}
{ 公式四：【竞价爆量抢筹副图指标】（技术指标公式 - 用于日线/分时看盘与盯盘）   }
{ 用途：在个股分时图或日K线副图中，实时量化显示今日竞价强度与抢筹买入信号       }
{------------------------------------------------------------------------------}

竞价金额_万: DYNAINFO(15) / 10000, NODRAW, COLORWHITE;
竞价占昨比: (DYNAINFO(15) / REF(AMOUNT, 1)) * 100, NODRAW, COLORYELLOW;
竞价量比: DYNAINFO(17), NODRAW, COLORCYAN;
竞价涨幅: (OPEN - REF(CLOSE, 1)) / REF(CLOSE, 1) * 100, NODRAW, COLORRED;

{ 警戒线参考 }
爆量基准线: 5.0, COLORGRAY, DOTLINE;
强势抢筹线: 8.0, COLORRED, DOTLINE;

{ 副图柱状图展示：根据竞价占昨比高低变色 }
STICKLINE(竞价占昨比 < 3.0, 0, 竞价占昨比, 3, 0), COLORGRAY;
STICKLINE(竞价占昨比 >= 3.0 AND 竞价占昨比 < 6.0, 0, 竞价占昨比, 3, 0), COLORYELLOW;
STICKLINE(竞价占昨比 >= 6.0 AND 竞价涨幅 > 0, 0, 竞价占昨比, 3, 0), COLORRED;
STICKLINE(竞价占昨比 >= 6.0 AND 竞价涨幅 <= 0, 0, 竞价占昨比, 3, 0), COLORGREEN;

{ 信号提示文字 }
DRAWTEXT(竞价占昨比 >= 6.0 AND 竞价涨幅 >= 2.0 AND 竞价涨幅 <= 7.0, 竞价占昨比 * 1.1, '★真龙抢筹'), COLORRED;
DRAWTEXT(竞价涨幅 <= 0.0 AND 竞价占昨比 >= 5.0, 竞价占昨比 * 1.1, '⚠核按钮分歧'), COLORGREEN;
